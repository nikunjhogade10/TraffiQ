00100: // Helper to generate polygon circle coordinates for sensor zone (~25m radius)
00101: function createGeoJsonCircle(center: [number, number], radiusInMeters: number, points = 32) {
00102:   const coords: [number, number][] = [];
00103:   const distanceX = radiusInMeters / (111320 * Math.cos((center[1] * Math.PI) / 180));
00104:   const distanceY = radiusInMeters / 110540;
00105: 
00106:   for (let i = 0; i < points; i++) {
00107:     const theta = (i / points) * (2 * Math.PI);
00108:     const x = distanceX * Math.cos(theta);
00109:     const y = distanceY * Math.sin(theta);
00110:     coords.push([center[0] + x, center[1] + y]);
00111:   }
00112:   coords.push(coords[0]);
00113:   return coords;
00114: }
